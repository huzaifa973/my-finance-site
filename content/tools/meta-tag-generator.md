<script>
(function(){
  var shell = document.getElementById('mtgShell');
  ToolPro.themeToggle(shell, document.getElementById('mtgTheme'));

  var f = {
    title: document.getElementById('mtgTitle'),
    desc: document.getElementById('mtgDesc'),
    keywords: document.getElementById('mtgKeywords'),
    canonical: document.getElementById('mtgCanonical'),
    ogImage: document.getElementById('mtgOgImage'),
    robots: document.getElementById('mtgRobots'),
    twCard: document.getElementById('mtgTwCard'),
    ogType: document.getElementById('mtgOgType'),
    togOg: document.getElementById('mtgTogOg'),
    togTw: document.getElementById('mtgTogTw')
  };
  var btn = document.getElementById('mtgBtn');
  var copyBtn = document.getElementById('mtgCopy');
  var dlBtn = document.getElementById('mtgDl');
  var presets = document.getElementById('mtgPresets');

  function esc(s){
    return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
  }

  function countColor(len, lo, hi){
    if(len === 0) return '';
    if(len < lo) return 'color:#b26a00;';
    if(len > hi) return 'color:#c62828;';
    return 'color:#157347;font-weight:800;';
  }

  function liveCounts(){
    var t = f.title.value.length, d = f.desc.value.length;
    var tc = document.getElementById('mtgTitleCount');
    var dc = document.getElementById('mtgDescCount');
    tc.textContent = t + ' chars'; tc.style.cssText = countColor(t, 50, 60);
    dc.textContent = d + ' chars'; dc.style.cssText = countColor(d, 150, 160);
    document.getElementById('mtgTitleHint').textContent =
      t === 0 ? 'Aim for 50–60 characters so the full title shows in Google.'
      : t < 50 ? 'A bit short — under 50 chars wastes your search-result space.'
      : t > 60 ? 'Too long — Google will cut the end off with an ellipsis.'
      : 'Perfect — this length shows fully in Google search results.';
    document.getElementById('mtgDescHint').textContent =
      d === 0 ? 'Aim for 150–160 characters — longer descriptions get cut off.'
      : d < 120 ? 'Quite short — add a call to action to earn the click.'
      : d > 160 ? 'Too long — Google truncates descriptions around 155–160 chars.'
      : 'Perfect — this length shows fully in Google search results.';
  }

  function refreshButtons(){
    btn.disabled = !(f.title.value.trim() && f.desc.value.trim());
  }

  ['title','desc','keywords','canonical','ogImage'].forEach(function(k){
    f[k].addEventListener('input', function(){ liveCounts(); refreshButtons(); });
  });

  presets.addEventListener('click', function(e){
    var chip = e.target.closest('.preset-chip'); if(!chip) return;
    presets.querySelectorAll('.preset-chip').forEach(function(c){ c.classList.remove('active'); });
    chip.classList.add('active');
    var p = chip.dataset.p;
    if(p === 'basic'){ f.togOg.checked = false; f.togTw.checked = false; }
    if(p === 'social'){ f.togOg.checked = true; f.togTw.checked = true; }
    if(p === 'full'){ f.togOg.checked = true; f.togTw.checked = true; }
    ToolPro.toast('Preset applied: ' + chip.textContent.trim());
  });

  function buildTags(){
    var t = f.title.value.trim(), d = f.desc.value.trim();
    var lines = [];
    lines.push('<!-- Primary meta tags -->');
    lines.push('<title>' + esc(t) + '</title>');
    lines.push('<meta name="title" content="' + esc(t) + '">');
    lines.push('<meta name="description" content="' + esc(d) + '">');
    var kw = f.keywords.value.trim();
    if(kw) lines.push('<meta name="keywords" content="' + esc(kw) + '">');
    var canon = f.canonical.value.trim();
    if(canon) lines.push('<link rel="canonical" href="' + esc(canon) + '">');
    lines.push('<meta name="robots" content="' + esc(f.robots.value) + '">');
    if(f.togOg.checked){
      lines.push('');
      lines.push('<!-- Open Graph / Facebook -->');
      lines.push('<meta property="og:type" content="' + esc(f.ogType.value) + '">');
      if(canon) lines.push('<meta property="og:url" content="' + esc(canon) + '">');
      lines.push('<meta property="og:title" content="' + esc(t) + '">');
      lines.push('<meta property="og:description" content="' + esc(d) + '">');
      var img = f.ogImage.value.trim();
      if(img){
        lines.push('<meta property="og:image" content="' + esc(img) + '">');
        lines.push('<meta property="og:image:width" content="1200">');
        lines.push('<meta property="og:image:height" content="630">');
      }
    }
    if(f.togTw.checked){
      lines.push('');
      lines.push('<!-- Twitter / X -->');
      lines.push('<meta name="twitter:card" content="' + esc(f.twCard.value) + '">');
      if(canon) lines.push('<meta name="twitter:url" content="' + esc(canon) + '">');
      lines.push('<meta name="twitter:title" content="' + esc(t) + '">');
      lines.push('<meta name="twitter:description" content="' + esc(d) + '">');
      var img2 = f.ogImage.value.trim();
      if(img2) lines.push('<meta name="twitter:image" content="' + esc(img2) + '">');
    }
    return lines.join('\n');
  }

  btn.addEventListener('click', function(){
    var progWrap = document.getElementById('mtgProgWrap');
    var progBar = document.getElementById('mtgProgBar');
    var resBox = document.getElementById('mtgResults');
    btn.disabled = true;
    progWrap.classList.add('show'); resBox.classList.remove('show');
    progBar.style.width = '40%';
    document.getElementById('mtgProgText').textContent = 'Building your tags…';
    setTimeout(function(){
      var html = buildTags();
      progBar.style.width = '100%';
      document.getElementById('mtgOut').value = html;
      var tl = f.title.value.length, dl = f.desc.value.length;
      var verdict = tl >= 50 && tl <= 60 ? '✅ Title length is in the sweet spot.'
        : '⚠️ Title is ' + tl + ' chars (aim for 50–60).';
      verdict += dl >= 120 && dl <= 160 ? ' ✅ Description length is good.'
        : ' ⚠️ Description is ' + dl + ' chars (aim for 150–160).';
      document.getElementById('mtgSummary').textContent =
        '🎉 ' + html.split('\n').filter(function(l){ return l.indexOf('<') === 0; }).length +
        ' tags generated. ' + verdict;
      progWrap.classList.remove('show');
      resBox.classList.add('show');
      resBox.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
      btn.disabled = false;
      copyBtn.disabled = false; dlBtn.disabled = false;
      ToolPro.toast('Meta tags generated', 'ok');
    }, 350);
  });

  copyBtn.addEventListener('click', function(){
    ToolPro.copyText(document.getElementById('mtgOut').value, 'Meta tags copied to clipboard');
  });

  dlBtn.addEventListener('click', function(){
    var url = 'data:text/html;charset=utf-8,' + encodeURIComponent(document.getElementById('mtgOut').value);
    ToolPro.download(url, 'meta-tags.html');
    ToolPro.toast('meta-tags.html downloaded', 'ok');
  });

  liveCounts();
  refreshButtons();
})();
</script>

## How it works

1. **Enter your page details** — title, description, keywords, canonical URL and a social share image. Live character guidance turns green when your lengths hit the proven ranges.
2. **Choose your tag options** — presets cover basic SEO, social sharing or the full pack; pick robots and Twitter-card behavior and toggle Open Graph/Twitter tags on or off.
3. **Generate, copy or download** — the tool builds clean, properly escaped HTML you can paste straight into your page's `<head>` section.

Everything runs in your browser — your text never leaves this page.

## Everyday uses

- **Bloggers & site owners** — nail the two fields Google actually shows (title + description) for every post you publish.
- **Freelancers** — ship client sites with professional, complete head tags instead of a bare `<title>`.
- **Marketers** — make sure shared links render with the right image and preview text on Facebook, LinkedIn and X.
- **Developers** — generate boilerplate `<head>` HTML in seconds when scaffolding new pages.
- **SEOs** — audit and rewrite underperforming titles and descriptions against the live character guidance.

**Rule of thumb:** write the description for humans first and search engines second — a clear, benefit-led description with a call to action beats keyword-stuffed copy every time.
