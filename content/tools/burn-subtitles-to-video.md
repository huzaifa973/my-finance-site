<script>
(function(){
  var shell = document.getElementById('bsShell');
  ToolPro.themeToggle(shell, document.getElementById('bsTheme'));

  var fontRange = document.getElementById('bsFontSize');
  var fontVal = document.getElementById('bsFontSizeVal');
  var posSel = document.getElementById('bsPosition');
  var marginRange = document.getElementById('bsMargin');
  var marginVal = document.getElementById('bsMarginVal');
  var boxToggle = document.getElementById('bsBoxToggle');
  var boxOn = true;
  var presets = document.getElementById('bsPresets');

  ToolPro.bindSlider(fontRange, fontVal, function(v){ return v + 'px'; });
  ToolPro.bindSlider(marginRange, marginVal, function(v){ return v + 'px'; });

  boxToggle.addEventListener('click', function(){
    boxOn = !boxOn;
    boxToggle.classList.toggle('on', boxOn);
    ToolPro.toast('Background box ' + (boxOn ? 'on' : 'off'));
  });
  boxToggle.classList.add('on');

  presets.addEventListener('click', function(e){
    var chip = e.target.closest('.preset-chip'); if(!chip) return;
    presets.querySelectorAll('.preset-chip').forEach(function(c){ c.classList.remove('active'); });
    chip.classList.add('active');
    fontRange.value = chip.dataset.size; fontRange.dispatchEvent(new Event('input'));
    posSel.value = chip.dataset.pos;
    boxOn = chip.dataset.box === '1';
    boxToggle.classList.toggle('on', boxOn);
    ToolPro.toast('Style preset applied: ' + chip.textContent.trim().split(' ')[0]);
  });

  // Validate the SRT text client-side so bad files fail fast with a clear message.
  function validateSrt(text){
    var blocks = text.trim().split(/\r?\n\s*\r?\n/);
    var cues = 0;
    var timeRe = /(\d{1,2}):(\d{2}):(\d{2})[,.](\d{2,3})\s*-->\s*(\d{1,2}):(\d{2}):(\d{2})[,.](\d{2,3})/;
    blocks.forEach(function(b){
      var lines = b.split(/\r?\n/);
      if(lines.length >= 2 && timeRe.test(lines[0] + ' ' + lines[1])) cues++;
    });
    return cues;
  }

  function buildForceStyle(){
    var size = parseInt(fontRange.value, 10);
    var margin = parseInt(marginRange.value, 10);
    var align = posSel.value === 'top' ? 8 : 2;
    // BorderStyle=4 draws a solid background box behind the text (ASS style).
    return 'FontName=Arial,FontSize=' + size +
      ',PrimaryColour=&H00FFFFFF,BackColour=&H90000000' +
      ',BorderStyle=' + (boxOn ? '4' : '1') +
      ',Outline=1,Shadow=0,Alignment=' + align + ',MarginV=' + margin;
  }

  VideoFX.init({
    prefix: 'bs',
    fxName: 'subtitled',
    needsSecond: true,
    buildArgs: function(ctx){
      // Write the subtitle text deterministically so the filter can find it.
      var vf = 'subtitles=' + ctx.second.fsName + ':force_style=' + buildForceStyle();
      return ['-i', ctx.main.fsName, '-vf', vf,
        '-c:v', 'libx264', '-preset', 'veryfast', '-crf', '23',
        '-c:a', 'aac', '-b:a', '160k', 'burned.mp4'];
    },
    onDuration: function(){}
  });

  // Live SRT validation + hint for the subtitle drop zone.
  var srtInput = document.getElementById('bsFile2');
  var srtDrop = document.getElementById('bsSrtDrop');
  var srtInfo = document.getElementById('bsSrtInfo');
  srtDrop.addEventListener('click', function(){ srtInput.click(); });
  srtDrop.addEventListener('dragover', function(e){ e.preventDefault(); srtDrop.classList.add('drag'); });
  srtDrop.addEventListener('dragleave', function(){ srtDrop.classList.remove('drag'); });
  srtDrop.addEventListener('drop', function(e){
    e.preventDefault(); srtDrop.classList.remove('drag');
    if(e.dataTransfer.files && e.dataTransfer.files.length){
      srtInput.files = e.dataTransfer.files;
      srtInput.dispatchEvent(new Event('change'));
    }
  });
  srtInput.addEventListener('change', function(){
    if(!srtInput.files.length) return;
    var f = srtInput.files[0];
    var reader = new FileReader();
    reader.onload = function(){
      var cues = validateSrt(String(reader.result || ''));
      if(cues > 0){
        srtInfo.style.display = '';
        srtInfo.innerHTML = '✅ Subtitle file looks good — <b>' + cues + '</b> caption cue' +
          (cues === 1 ? '' : 's') + ' found.';
        ToolPro.toast(cues + ' caption cues loaded', 'ok');
      } else {
        srtInfo.style.display = '';
        srtInfo.textContent = '⚠️ That file does not look like a valid .srt subtitle file — captions may not burn correctly.';
        ToolPro.toast('Subtitle file may be invalid', 'err');
      }
    };
    reader.readAsText(f);
  });
})();

## How it works

1. **Add your video** — drag it in or browse. It stays on your device; nothing uploads.
2. **Add your .srt file** — we validate it instantly and count the caption cues for you.
3. **Pick a style** — font size, top/bottom position, edge margin, and a background box for readability.
4. **Burn** — the subtitle text is permanently drawn into every frame with ffmpeg, right in your browser, and you download the finished MP4.

## Everyday uses

- **Creators** — most social viewers watch muted; burned-in captions keep them watching.
- **Educators** — make lectures and tutorials accessible without a subtitle player.
- **Course sellers** — hard-coded captions work on any platform, no file-format juggling.
- **Interviewers** — burn translated captions into testimonial and interview clips.
- **Marketers** — captions increase watch time and message retention on ads.

**Privacy:** your video and subtitle files never leave your browser — the burn happens locally on your device.
